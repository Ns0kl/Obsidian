# 定数と列挙型

## 概要

**定数**と**列挙型**は、固定値を安全に管理する方法です。TypeScriptではEnumやconst assertionを活用します。

## const 宣言

### 基本的な定数

```typescript
// プリミティブ型
const MAX_USERS: number = 100;
const API_URL: string = "https://api.example.com";
const DEFAULT_TIMEOUT: number = 5000;

// 複合型
const DATABASE_CONFIG: { host: string; port: number } = {
    host: "localhost",
    port: 5432
};

// 再代入はできない
// MAX_USERS = 200;  // Error
```

### const オブジェクト

```typescript
const config = {
    apiUrl: "https://api.example.com",
    timeout: 5000
} as const;

// 値が固定される
type ConfigType = typeof config;
// {
//     readonly apiUrl: "https://api.example.com";
//     readonly timeout: 5000;
// }
```

## Enum（列挙型）

### 数値 Enum

```typescript
enum Status {
    Pending = 0,
    Active = 1,
    Inactive = 2
}

const userStatus: Status = Status.Active;
console.log(userStatus);  // 1
console.log(Status[1]);   // "Active"（逆マッピング）

// デフォルトは0から始まる
enum Color {
    Red,      // 0
    Green,    // 1
    Blue      // 2
}
```

### 文字列 Enum

```typescript
enum Direction {
    Up = "UP",
    Down = "DOWN",
    Left = "LEFT",
    Right = "RIGHT"
}

const move: Direction = Direction.Up;
console.log(move);  // "UP"

// if文で使用
if (move === Direction.Up) {
    console.log("移動：上");
}
```

### 異種 Enum

```typescript
enum Mixed {
    No = 0,
    Yes = "YES"
}

console.log(Mixed.No);    // 0
console.log(Mixed.Yes);   // "YES"
```

## const Assertion

```typescript
// as const で型を厳密に
const priorities = {
    low: "low",
    medium: "medium",
    high: "high"
} as const;

type Priority = typeof priorities[keyof typeof priorities];
// "low" | "medium" | "high"

// 配列
const statuses = ["pending", "active", "inactive"] as const;
type Status = typeof statuses[number];
// "pending" | "active" | "inactive"
```

## ユーティリティ型

### ### Partial< T >

```typescript
interface User {
    id: number;
    name: string;
    email: string;
}

// すべてのプロパティをオプショナルに
type PartialUser = Partial<User>;
// { id?: number; name?: string; email?: string; }

function updateUser(
    id: number,
    updates: PartialUser
): User {
    return {
        id,
        name: updates.name || "Unknown",
        email: updates.email || ""
    };
}

updateUser(1, { name: "Alice" });
```

### Required< T >

```typescript
interface Config {
    apiUrl?: string;
    timeout?: number;
}

// すべてのプロパティを必須に
type RequiredConfig = Required<Config>;
// { apiUrl: string; timeout: number; }

const config: RequiredConfig = {
    apiUrl: "https://api.example.com",
    timeout: 5000
};
```

### Record<K, T>

```typescript
// キーと値の型を指定
type Status = "pending" | "active" | "inactive";

const statusMessages: Record<Status, string> = {
    pending: "処理待ち",
    active: "実行中",
    inactive: "非アクティブ"
};

console.log(statusMessages.active);  // "実行中"

// より複雑な例
type Permission = "read" | "write" | "delete";
type Role = "admin" | "user" | "guest";

const permissions: Record<Role, Permission[]> = {
    admin: ["read", "write", "delete"],
    user: ["read", "write"],
    guest: ["read"]
};
```

### Pick<T, K>

```typescript
interface User {
    id: number;
    name: string;
    email: string;
    phone: string;
}

// 指定されたプロパティだけを選択
type UserPreview = Pick<User, "name" | "email">;
// { name: string; email: string; }

const preview: UserPreview = {
    name: "Alice",
    email: "alice@example.com"
};
```

### Omit<T, K>

```typescript
// 指定されたプロパティを除外
type UserWithoutId = Omit<User, "id">;
// { name: string; email: string; phone: string; }

const user: UserWithoutId = {
    name: "Bob",
    email: "bob@example.com",
    phone: "123-456-7890"
};
```

### Exclude<T, U>

```typescript
type Status = "pending" | "active" | "inactive" | "deleted";

// "deleted"を除外
type ActiveStatus = Exclude<Status, "deleted">;
// "pending" | "active" | "inactive"

const currentStatus: ActiveStatus = "active";
```

### Extract<T, U>

```typescript
// 交集合を取得
type HttpMethod = "GET" | "POST" | "PUT" | "DELETE";
type MutatingMethod = "POST" | "PUT" | "DELETE" | "PATCH";

type RequestMethod = Extract<HttpMethod, MutatingMethod>;
// "POST" | "PUT" | "DELETE"
```

### ReturnType< T >

```typescript
function getUserData(id: number): { id: number; name: string } {
    return { id, name: "Alice" };
}

// 関数の戻り値型を取得
type UserData = ReturnType<typeof getUserData>;
// { id: number; name: string; }
```

### Parameters< T >

```typescript
function sendEmail(
    to: string,
    subject: string,
    body: string
): void {
    // 実装
}

// 関数のパラメータ型を取得
type EmailParams = Parameters<typeof sendEmail>;
// [to: string, subject: string, body: string]

const args: EmailParams = ["user@example.com", "Title", "Content"];
```

## 実用例

### API レスポンス型管理

```typescript
interface ApiResponse<T> {
    status: number;
    data: T;
    meta?: {
        total: number;
        page: number;
    };
}

// ユーザー固有のレスポンス型
interface User {
    id: number;
    name: string;
    email: string;
}

type UserResponse = ApiResponse<User>;
type UserListResponse = ApiResponse<User[]>;

async function getUser(id: number): Promise<UserResponse> {
    const response = await fetch(`/api/users/${id}`);
    return response.json();
}
```

### ステータス管理

```typescript
enum OrderStatus {
    Pending = "PENDING",
    Processing = "PROCESSING",
    Shipped = "SHIPPED",
    Delivered = "DELIVERED",
    Cancelled = "CANCELLED"
}

// 順序状態の説明
const statusDescriptions: Record<OrderStatus, string> = {
    [OrderStatus.Pending]: "注文待機中",
    [OrderStatus.Processing]: "処理中",
    [OrderStatus.Shipped]: "配送中",
    [OrderStatus.Delivered]: "配送完了",
    [OrderStatus.Cancelled]: "キャンセル"
};

// 許可された遷移
type StatusTransition = Record<
    OrderStatus,
    OrderStatus[]
>;

const allowedTransitions: StatusTransition = {
    [OrderStatus.Pending]: [OrderStatus.Processing, OrderStatus.Cancelled],
    [OrderStatus.Processing]: [OrderStatus.Shipped, OrderStatus.Cancelled],
    [OrderStatus.Shipped]: [OrderStatus.Delivered],
    [OrderStatus.Delivered]: [],
    [OrderStatus.Cancelled]: []
};
```

### 権限管理

```typescript
type Permission = 
    | "user:read"
    | "user:write"
    | "user:delete"
    | "post:read"
    | "post:write"
    | "post:delete";

type Role = "admin" | "editor" | "viewer";

const rolePermissions: Record<Role, Permission[]> = {
    admin: [
        "user:read", "user:write", "user:delete",
        "post:read", "post:write", "post:delete"
    ],
    editor: [
        "user:read",
        "post:read", "post:write", "post:delete"
    ],
    viewer: [
        "user:read",
        "post:read"
    ]
};

function hasPermission(
    role: Role,
    permission: Permission
): boolean {
    return rolePermissions[role].includes(permission);
}

console.log(hasPermission("editor", "post:write"));  // true
console.log(hasPermission("viewer", "user:write"));  // false
```

## 📌 重要なポイント

- **const**: 再代入不可の定数
- **as const**: 値を固定型にする
- **Enum**: 関連する定数をグループ化
- **Pick/Omit**: インターフェースから部分型を作成
- **Record**: キーと値の対応関係を型安全に管理
- **ユーティリティ型**: DRYな型定義を実現
