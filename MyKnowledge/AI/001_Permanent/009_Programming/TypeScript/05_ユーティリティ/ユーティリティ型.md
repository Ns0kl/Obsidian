# ユーティリティ型

## 概要

**ユーティリティ型**は、TypeScriptが提供する組み込み型です。既存の型を変換して新しい型を作成します。

## 基本的なユーティリティ型

### Partial<T>

```typescript
interface User {
    id: number;
    name: string;
    email: string;
}

// すべてのプロパティをオプショナルに
type PartialUser = Partial<User>;
// { id?: number; name?: string; email?: string; }

// 更新関数で使用
function updateUser(id: number, update: PartialUser): User {
    // 実装
    return { id, name: "Updated", email: "email@example.com" };
}

updateUser(1, { name: "Alice" });
updateUser(1, { email: "alice@example.com" });
updateUser(1, { name: "Alice", email: "alice@example.com" });
```

### Required<T>

```typescript
interface Config {
    apiUrl?: string;
    timeout?: number;
    retryCount?: number;
}

// すべてのプロパティを必須に
type RequiredConfig = Required<Config>;
// { apiUrl: string; timeout: number; retryCount: number; }

const config: RequiredConfig = {
    apiUrl: "https://api.example.com",
    timeout: 5000,
    retryCount: 3
};
```

### Readonly<T>

```typescript
interface Point {
    x: number;
    y: number;
}

// すべてのプロパティを読み取り専用に
type ReadonlyPoint = Readonly<Point>;
// { readonly x: number; readonly y: number; }

const point: ReadonlyPoint = { x: 0, y: 0 };
// point.x = 10;  // Error: readonly
```

### Record<K, T>

```typescript
// キーと値の型を指定
type Status = "active" | "inactive" | "pending";
type StatusInfo = {
    label: string;
    color: string;
};

const statusMap: Record<Status, StatusInfo> = {
    active: { label: "アクティブ", color: "green" },
    inactive: { label: "非アクティブ", color: "gray" },
    pending: { label: "保留中", color: "yellow" }
};

console.log(statusMap.active.label);  // "アクティブ"

// 数値キー
type RoleId = 1 | 2 | 3;
type RoleName = Record<RoleId, string>;

const roles: RoleName = {
    1: "Admin",
    2: "User",
    3: "Guest"
};
```

### Pick<T, K>

```typescript
interface User {
    id: number;
    name: string;
    email: string;
    password: string;
    phone: string;
}

// 指定されたプロパティだけを選択
type UserPreview = Pick<User, "name" | "email">;
// { name: string; email: string; }

type UserPublic = Pick<User, "id" | "name">;
// { id: number; name: string; }

const preview: UserPreview = {
    name: "Alice",
    email: "alice@example.com"
};
```

### Omit<T, K>

```typescript
// 指定されたプロパティを除外
type UserWithoutPassword = Omit<User, "password">;
// { id: number; name: string; email: string; phone: string; }

type UserWithoutId = Omit<User, "id" | "password">;
// { name: string; email: string; phone: string; }

const user: UserWithoutPassword = {
    id: 1,
    name: "Bob",
    email: "bob@example.com",
    phone: "123-456-7890"
};
```

## 型変換ユーティリティ

### Exclude<T, U>

```typescript
type Status = "pending" | "active" | "inactive" | "deleted";

// "deleted"を除外
type ActiveStatus = Exclude<Status, "deleted">;
// "pending" | "active" | "inactive"

// 複数除外
type ProcessingStatus = Exclude<Status, "deleted" | "inactive">;
// "pending" | "active"

const status: ActiveStatus = "active";
```

### Extract<T, U>

```typescript
type Status = "pending" | "active" | "inactive" | "deleted";
type ProcessingStatuses = "pending" | "active" | "processing";

// 交集合を取得
type CommonStatus = Extract<Status, ProcessingStatuses>;
// "pending" | "active"

// 条件付き型でも使用可
type StringOrNumber = string | number | boolean;
type OnlyStrings = Extract<StringOrNumber, string>;
// string
```

### NonNullable<T>

```typescript
type MaybeUser = User | null | undefined;

// null と undefined を除外
type DefiniteUser = NonNullable<MaybeUser>;
// User

// ユーティリティ関数
function unwrap<T>(value: T | null | undefined): NonNullable<T> {
    if (value === null || value === undefined) {
        throw new Error("Value is null or undefined");
    }
    return value;
}

const maybeString: string | null = "hello";
const definiteString: string = unwrap(maybeString);
```

## 関数型ユーティリティ

### ReturnType<T>

```typescript
function getUserData(id: number): { id: number; name: string } {
    return { id, name: "Alice" };
}

// 関数の戻り値型を取得
type UserData = ReturnType<typeof getUserData>;
// { id: number; name: string; }

const userData: UserData = { id: 1, name: "Bob" };
```

### Parameters<T>

```typescript
function sendEmail(
    to: string,
    subject: string,
    body: string
): Promise<void> {
    // 実装
    return Promise.resolve();
}

// 関数のパラメータ型をタプルで取得
type EmailParams = Parameters<typeof sendEmail>;
// [to: string, subject: string, body: string]

function logEmailSending(...args: EmailParams): void {
    const [to, subject, body] = args;
    console.log(`Sending email to ${to}: ${subject}`);
}

logEmailSending("user@example.com", "Hello", "Hi there!");
```

### ConstructorParameters<T>

```typescript
class User {
    constructor(public name: string, public age: number) {}
}

// コンストラクタのパラメータ型を取得
type UserConstructorParams = ConstructorParameters<typeof User>;
// [name: string, age: number]

const params: UserConstructorParams = ["Alice", 30];
const user = new User(...params);
```

### InstanceType<T>

```typescript
class Dog {
    bark(): void {
        console.log("Woof!");
    }
}

// コンストラクタ型からインスタンス型を取得
type DogInstance = InstanceType<typeof Dog>;
// Dog

const dog: DogInstance = new Dog();
dog.bark();
```

### ThisParameterType<T>・OmitThisParameter<T>

```typescript
interface Logger {
    log(this: Logger, message: string): void;
}

// thisパラメータの型を取得
type LoggerThis = ThisParameterType<Logger["log"]>;
// Logger

// thisパラメータを除いた関数型
type PlainLog = OmitThisParameter<Logger["log"]>;
// (message: string) => void
```

## 条件付き型

### 基本形

```typescript
// Tが string の場合は string、そうでなければ number
type StringOrNumber<T> = T extends string ? string : number;

type A = StringOrNumber<"hello">;      // string
type B = StringOrNumber<123>;          // number

// 複数の条件
type FlexibleReturnType<T> = T extends string
    ? string
    : T extends number
    ? number
    : boolean;

type X = FlexibleReturnType<"text">;   // string
type Y = FlexibleReturnType<42>;       // number
type Z = FlexibleReturnType<true>;     // boolean
```

### Infer キーワード

```typescript
// 配列の要素型を取得
type ArrayElement<T> = T extends (infer U)[] ? U : never;

type StringArray = ArrayElement<string[]>;  // string
type NumberArray = ArrayElement<number[]>; // number

// 関数の戻り値型を取得
type FunctionReturn<T> = T extends (...args: any[]) => infer R ? R : never;

const fn = (x: number) => x.toString();
type ReturnOfFn = FunctionReturn<typeof fn>;  // string

// ユーザーのプロパティから値型を取得
type PropertyValue<T, K extends keyof T> = T[K] extends infer V
    ? V
    : never;

type NameType = PropertyValue<User, "name">;  // string
```

## 高度なパターン

### Getters と Setters

```typescript
// ゲッターから型を取得
type Getter<T> = {
    [K in keyof T]: T[K] extends (...args: any[]) => infer R ? R : never;
};

// セッターから型を取得
type Setter<T> = {
    [K in keyof T]: T[K] extends (value: infer V) => any ? V : never;
};
```

### デコレータと互換性のある型

```typescript
type Decorator<T extends Function> = (
    target: any,
    propertyKey: string | symbol | undefined,
    descriptor: PropertyDescriptor
) => PropertyDescriptor;

// クラスメソッドのジェネリック型
type MethodDecorator<T extends (...args: any[]) => any> = (
    target: any,
    propertyKey: string | symbol,
    descriptor: TypedPropertyDescriptor<T>
) => TypedPropertyDescriptor<T> | void;
```

## 実用例

```typescript
// API レスポンスの共通部分
interface ApiResponse<T> {
    success: boolean;
    data: T;
    timestamp: Date;
}

type ApiDataOnly<T> = T extends ApiResponse<infer U> ? U : never;

interface User {
    id: number;
    name: string;
}

type UserFromResponse = ApiDataOnly<ApiResponse<User>>;
// { id: number; name: string; }

// フォーム検証結果
type FormErrors<T> = Partial<Record<keyof T, string[]>>;

type UserFormErrors = FormErrors<User>;
// {
//     id?: string[];
//     name?: string[];
// }
```

## 📌 重要なポイント

- **Partial/Required**: プロパティの必須性を変更
- **Pick/Omit**: プロパティの部分集合を作成
- **Record**: キーと値の対応を型安全に
- **Exclude/Extract**: ユニオン型から型を抽出
- **ReturnType/Parameters**: 関数の型から情報を抽出
- **条件付き型**: 型を条件分岐で選択
- **Infer**: 型から値を推論
