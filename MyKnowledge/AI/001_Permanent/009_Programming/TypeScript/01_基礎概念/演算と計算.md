# 演算と計算

## 概要

TypeScriptの演算子と計算方法について学びます。JavaScriptとほぼ同じですが、型安全性が強化されています。

## 算術演算子

```typescript
// 加算
const result1: number = 9 + 2;      // 11

// 減算
const result2: number = 9 - 2;      // 7

// 乗算
const result3: number = 9 * 2;      // 18

// 除算
const result4: number = 9 / 2;      // 4.5

// 剰余算
const result5: number = 9 % 2;      // 1

// 累乗
const result6: number = 9 ** 2;     // 81

// インクリメント・デクリメント
let x: number = 5;
x++;                                 // 6
x--;                                 // 5
++x;                                 // 6（前置）
x++;                                 // 7（後置）
```

## 比較演算子

```typescript
const x: number = 10;
const y: number = 20;

// 等値比較
console.log(x == y);    // false（値の比較）
console.log(x === y);   // false（値と型の比較、推奨）

// 不等号
console.log(x != y);    // true
console.log(x !== y);   // true（型も考慮）

// 大小比較
console.log(x < y);     // true
console.log(x <= y);    // true
console.log(x > y);     // false
console.log(x >= y);    // false
```

## 論理演算子

```typescript
// AND演算子：両方trueのときtrue
console.log(true && true);      // true
console.log(true && false);     // false

// OR演算子：どちらかtrueのときtrue
console.log(true || false);     // true
console.log(false || false);    // false

// NOT演算子：逆にする
console.log(!true);             // false
console.log(!false);            // true

// 短絡評価
let value: number | undefined;
value = value || 10;  // valueがfalsy（undefined）なら10を代入
```

## 文字列演算

### 連結

```typescript
const name: string = "Alice";
const greeting: string = "Hello, " + name;
console.log(greeting);  // "Hello, Alice"

// 複数行連結
const long: string = 
    "This is a " +
    "very long " +
    "string";
```

### テンプレートリテラル

```typescript
const name: string = "Bob";
const age: number = 30;
const city: string = "Tokyo";

// バックティックで囲む
const message: string = `${name} is ${age} years old from ${city}`;
console.log(message);  // "Bob is 30 years old from Tokyo"

// 複数行
const multiLine: string = `
    First line
    Second line
    Third line
`;

// 式の埋め込み
const result: string = `The answer is ${10 + 20}`;
console.log(result);  // "The answer is 30"
```

## 代入演算子

```typescript
let x: number = 10;

// 複合代入演算子
x += 5;     // x = x + 5   → 15
x -= 3;     // x = x - 3   → 12
x *= 2;     // x = x * 2   → 24
x /= 4;     // x = x / 4   → 6
x %= 5;     // x = x % 5   → 1

// 文字列連結
let str: string = "Hello";
str += " World";
console.log(str);  // "Hello World"
```

## 条件演算子（三項演算子）

```typescript
const age: number = 20;
const status: string = age >= 18 ? "成人" : "未成年";
console.log(status);  // "成人"

// ネスト
const score: number = 85;
const grade: string = 
    score >= 90 ? "A" :
    score >= 80 ? "B" :
    score >= 70 ? "C" : "F";
console.log(grade);  // "B"
```

## Null合体演算子

```typescript
// ??：左がnullまたはundefinedなら右を使用
let value: string | null = null;
let result: string = value ?? "default";
console.log(result);  // "default"

// 0やfalseは考慮しない（||と異なる）
let count: number = 0;
let total: number = count ?? 10;
console.log(total);  // 0（||なら10になる）
```

## オプショナルチェーニング

```typescript
interface User {
    name: string;
    address?: {
        city: string;
        zip: string;
    };
}

const user: User = { name: "Alice" };

// 従来の方法
if (user.address && user.address.city) {
    console.log(user.address.city);
}

// オプショナルチェーニング
console.log(user.address?.city);  // undefined（エラーにならない）
console.log(user.address?.city ?? "Unknown");  // "Unknown"
```

## 型ガード

```typescript
function process(input: string | number): void {
    // typeof を使った型ガード
    if (typeof input === 'string') {
        console.log(input.toUpperCase());  // string型のメソッド
    } else {
        console.log(input * 2);            // number型の演算
    }
}

// in演算子
interface Bird {
    fly(): void;
}

interface Fish {
    swim(): void;
}

function move(animal: Bird | Fish): void {
    if ('fly' in animal) {
        animal.fly();
    } else {
        animal.swim();
    }
}
```

## 📌 重要なポイント

- **===を優先**: 型を考慮した比較演算子を使用
- **テンプレートリテラル**: 文字列連結に便利
- **短絡評価**: &&と||を活用
- **オプショナルチェーニング**: nullチェックを簡潔に
- **Null合体演算子**: nullやundefinedのデフォルト値処理
