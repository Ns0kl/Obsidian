# 制御文

## 概要

制御文は、プログラムの実行フローを制御します。条件によって異なるコードを実行したり、繰り返し処理を行います。

## if・else文

### 基本形

```typescript
const age: number = 18;

if (age >= 20) {
    console.log("成人です");
} else if (age >= 18) {
    console.log("18歳以上です");
} else {
    console.log("未成年です");
}
// 出力: 18歳以上です
```

### 複合条件

```typescript
const score: number = 85;
const attendance: number = 90;

if (score >= 80 && attendance >= 80) {
    console.log("合格です");
} else if (score >= 70 || attendance >= 90) {
    console.log("条件付き合格です");
} else {
    console.log("不合格です");
}
```

## switch文

```typescript
const day: number = 3;
let dayName: string;

switch (day) {
    case 1:
        dayName = "Monday";
        break;
    case 2:
        dayName = "Tuesday";
        break;
    case 3:
        dayName = "Wednesday";
        break;
    default:
        dayName = "Unknown";
}

console.log(dayName);  // "Wednesday"
```

### switch with型ガード

```typescript
type Status = "pending" | "approved" | "rejected";

function handleStatus(status: Status): void {
    switch (status) {
        case "pending":
            console.log("待機中...");
            break;
        case "approved":
            console.log("承認されました");
            break;
        case "rejected":
            console.log("却下されました");
            break;
    }
}
```

## for文

### 基本形

```typescript
// C形式のfor文
for (let i: number = 0; i < 5; i++) {
    console.log(i);  // 0, 1, 2, 3, 4
}

// ステップを指定
for (let i: number = 0; i < 10; i += 2) {
    console.log(i);  // 0, 2, 4, 6, 8
}
```

### for...of ループ

```typescript
const fruits: string[] = ["apple", "banana", "orange"];

for (const fruit of fruits) {
    console.log(fruit);  // apple, banana, orange
}

// インデックス付き
for (const [index, fruit] of fruits.entries()) {
    console.log(`${index}: ${fruit}`);
}
```

### for...in ループ（オブジェクト）

```typescript
const person: { [key: string]: string } = {
    name: "Alice",
    city: "Tokyo",
    hobby: "Reading"
};

for (const key in person) {
    console.log(`${key}: ${person[key]}`);
}
```

## while・do...while文

```typescript
// while文
let count: number = 0;
while (count < 5) {
    console.log(count);  // 0, 1, 2, 3, 4
    count++;
}

// do...while文（最低1回実行）
let i: number = 0;
do {
    console.log(i);
    i++;
} while (i < 3);  // 0, 1, 2
```

## break・continue文

### break文

```typescript
// ループを途中で終了
for (let i: number = 0; i < 10; i++) {
    if (i === 5) {
        break;  // ループを抜ける
    }
    console.log(i);  // 0, 1, 2, 3, 4
}

// ラベル付きbreak（ネストされたループから抜ける）
outerLoop:
for (let i: number = 0; i < 3; i++) {
    for (let j: number = 0; j < 3; j++) {
        if (i === 1 && j === 1) {
            break outerLoop;  // 外側のループも抜ける
        }
        console.log(`${i},${j}`);
    }
}
```

### continue文

```typescript
// 現在の反復をスキップ
for (let i: number = 0; i < 5; i++) {
    if (i === 2) {
        continue;  // i=2をスキップ
    }
    console.log(i);  // 0, 1, 3, 4
}
```

## ネストされた制御文

```typescript
for (let i: number = 1; i <= 3; i++) {
    for (let j: number = 1; j <= 3; j++) {
        console.log(`${i} * ${j} = ${i * j}`);
    }
}

// 九九表
for (let i: number = 1; i <= 9; i++) {
    for (let j: number = 1; j <= 9; j++) {
        if (j > 1) process.stdout.write(" ");
        process.stdout.write(String(i * j).padStart(2));
    }
    console.log();
}
```

## 高階関数による制御

### forEach

```typescript
const numbers: number[] = [1, 2, 3, 4, 5];
numbers.forEach((num: number, index: number) => {
    console.log(`${index}: ${num}`);
});
```

### filter・map・reduce

```typescript
const numbers: number[] = [1, 2, 3, 4, 5];

// filter：条件に合う要素だけ
const evens: number[] = numbers.filter(n => n % 2 === 0);
console.log(evens);  // [2, 4]

// map：各要素を変換
const doubled: number[] = numbers.map(n => n * 2);
console.log(doubled);  // [2, 4, 6, 8, 10]

// reduce：値を集約
const sum: number = numbers.reduce((acc, n) => acc + n, 0);
console.log(sum);  // 15
```

## 📌 重要なポイント

- **if/else**: 条件分岐の基本
- **switch**: 複数の値に対応する場合に便利
- **for...of**: 配列の反復処理に推奨
- **for...in**: オブジェクトのキーを反復
- **break/continue**: ループの制御
- **高階関数**: 関数型プログラミングのスタイル
