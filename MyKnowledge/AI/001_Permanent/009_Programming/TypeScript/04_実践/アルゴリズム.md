# アルゴリズム

## 概要

**アルゴリズム**は、問題を解決する手順です。ソート、検索、グラフアルゴリズムなど、様々な重要なアルゴリズムがあります。

## ソートアルゴリズム

### バブルソート

```typescript
function bubbleSort(arr: number[]): number[] {
    const n: number = arr.length;
    
    for (let i: number = 0; i < n - 1; i++) {
        for (let j: number = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                // スワップ
                [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
            }
        }
    }
    
    return arr;
}

console.log(bubbleSort([64, 34, 25, 12, 22, 11, 90]));
// [11, 12, 22, 25, 34, 64, 90]
```

### クイックソート

```typescript
function quickSort(arr: number[]): number[] {
    if (arr.length <= 1) {
        return arr;
    }

    const pivot: number = arr[Math.floor(arr.length / 2)];
    const left: number[] = arr.filter(x => x < pivot);
    const middle: number[] = arr.filter(x => x === pivot);
    const right: number[] = arr.filter(x => x > pivot);

    return [...quickSort(left), ...middle, ...quickSort(right)];
}

console.log(quickSort([64, 34, 25, 12, 22, 11, 90]));
// [11, 12, 22, 25, 34, 64, 90]
```

### マージソート

```typescript
function mergeSort(arr: number[]): number[] {
    if (arr.length <= 1) {
        return arr;
    }

    const mid: number = Math.floor(arr.length / 2);
    const left: number[] = mergeSort(arr.slice(0, mid));
    const right: number[] = mergeSort(arr.slice(mid));

    return merge(left, right);
}

function merge(left: number[], right: number[]): number[] {
    const result: number[] = [];
    let i: number = 0;
    let j: number = 0;

    while (i < left.length && j < right.length) {
        if (left[i] <= right[j]) {
            result.push(left[i++]);
        } else {
            result.push(right[j++]);
        }
    }

    return [...result, ...left.slice(i), ...right.slice(j)];
}

console.log(mergeSort([64, 34, 25, 12, 22, 11, 90]));
// [11, 12, 22, 25, 34, 64, 90]
```

## 検索アルゴリズム

### 線形探索

```typescript
function linearSearch<T>(arr: T[], target: T): number {
    for (let i: number = 0; i < arr.length; i++) {
        if (arr[i] === target) {
            return i;
        }
    }
    return -1;
}

console.log(linearSearch([10, 20, 30, 40, 50], 30));  // 2
console.log(linearSearch([10, 20, 30, 40, 50], 100)); // -1
```

### 二分探索

```typescript
function binarySearch(arr: number[], target: number): number {
    let left: number = 0;
    let right: number = arr.length - 1;

    while (left <= right) {
        const mid: number = Math.floor((left + right) / 2);

        if (arr[mid] === target) {
            return mid;
        } else if (arr[mid] < target) {
            left = mid + 1;
        } else {
            right = mid - 1;
        }
    }

    return -1;
}

const sortedArr: number[] = [10, 20, 30, 40, 50, 60, 70];
console.log(binarySearch(sortedArr, 40));  // 3
console.log(binarySearch(sortedArr, 25));  // -1
```

## 数値アルゴリズム

### 最大公約数（GCD）

```typescript
// ユークリッド互除法
function gcd(a: number, b: number): number {
    while (b !== 0) {
        [a, b] = [b, a % b];
    }
    return a;
}

console.log(gcd(48, 18));  // 6
console.log(gcd(100, 50)); // 50
```

### 最小公倍数（LCM）

```typescript
function lcm(a: number, b: number): number {
    return Math.abs(a * b) / gcd(a, b);
}

console.log(lcm(12, 18));  // 36
console.log(lcm(4, 6));    // 12
```

### 素数判定

```typescript
function isPrime(n: number): boolean {
    if (n <= 1) return false;
    if (n <= 3) return true;
    if (n % 2 === 0 || n % 3 === 0) return false;

    for (let i: number = 5; i * i <= n; i += 6) {
        if (n % i === 0 || n % (i + 2) === 0) {
            return false;
        }
    }

    return true;
}

console.log(isPrime(2));   // true
console.log(isPrime(17));  // true
console.log(isPrime(20));  // false
```

### フィボナッチ数列

```typescript
// 再帰版（遅い）
function fibonacci(n: number): number {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

// 動的計画法（高速）
function fibonacciDP(n: number): number {
    if (n <= 1) return n;
    
    const dp: number[] = [0, 1];
    for (let i: number = 2; i <= n; i++) {
        dp[i] = dp[i - 1] + dp[i - 2];
    }
    
    return dp[n];
}

console.log(fibonacciDP(10));  // 55
```

## グラフアルゴリズム

### 深さ優先探索（DFS）

```typescript
function dfs(
    graph: Map<number, number[]>,
    start: number,
    visited: Set<number> = new Set()
): number[] {
    const result: number[] = [];
    
    if (visited.has(start)) return result;
    
    visited.add(start);
    result.push(start);

    const neighbors: number[] | undefined = graph.get(start);
    if (neighbors) {
        for (const neighbor of neighbors) {
            result.push(...dfs(graph, neighbor, visited));
        }
    }

    return result;
}

const graph: Map<number, number[]> = new Map([
    [1, [2, 3]],
    [2, [4]],
    [3, [4]],
    [4, []]
]);

console.log(dfs(graph, 1));  // [1, 2, 4, 3]
```

### 幅優先探索（BFS）

```typescript
function bfs(graph: Map<number, number[]>, start: number): number[] {
    const visited: Set<number> = new Set([start]);
    const queue: number[] = [start];
    const result: number[] = [];

    while (queue.length > 0) {
        const node: number = queue.shift()!;
        result.push(node);

        const neighbors: number[] | undefined = graph.get(node);
        if (neighbors) {
            for (const neighbor of neighbors) {
                if (!visited.has(neighbor)) {
                    visited.add(neighbor);
                    queue.push(neighbor);
                }
            }
        }
    }

    return result;
}

console.log(bfs(graph, 1));  // [1, 2, 3, 4]
```

## 動的計画法

### ナップサック問題

```typescript
function knapsack(
    weights: number[],
    values: number[],
    capacity: number
): number {
    const n: number = weights.length;
    const dp: number[][] = Array(n + 1).fill(null).map(() =>
        Array(capacity + 1).fill(0)
    );

    for (let i: number = 1; i <= n; i++) {
        for (let w: number = 1; w <= capacity; w++) {
            if (weights[i - 1] <= w) {
                dp[i][w] = Math.max(
                    values[i - 1] + dp[i - 1][w - weights[i - 1]],
                    dp[i - 1][w]
                );
            } else {
                dp[i][w] = dp[i - 1][w];
            }
        }
    }

    return dp[n][capacity];
}

const weights: number[] = [2, 3, 4, 5];
const values: number[] = [3, 4, 5, 6];
console.log(knapsack(weights, values, 8));  // 13
```

## 📌 重要なポイント

- **ソート**: 様々な実装方法がある（時間計算量が異なる）
- **検索**: 二分探索は整列済みデータで高速
- **GCD・LCM**: 数値計算の基本
- **動的計画法**: 部分問題の結果をキャッシュして高速化
