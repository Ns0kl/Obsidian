# 配列

## 概要

**配列**は、複数の要素を順序付きで保持するデータ構造です。TypeScriptでは要素の型を指定できます。

## 配列の基本

### 作成と型

```typescript
// 型推論
const numbers = [0, 1, 2, 3, 4, 5];
console.log(typeof numbers);  // object

// 明示的な型指定
const strings: string[] = ["apple", "banana", "cherry"];
const ages: Array<number> = [25, 30, 35];
const mixed: (string | number)[] = ["alice", 25, "bob", 30];

// 空の配列
const empty: number[] = [];
```

### 長さとアクセス

```typescript
const arr: number[] = [10, 20, 30, 40, 50];

console.log(arr.length);   // 5
console.log(arr[0]);       // 10
console.log(arr[4]);       // 50
console.log(arr[-1]);      // undefined（JavaScriptでは負のインデックスは非対応）
```

## 配列操作メソッド

### 要素の追加・削除

```typescript
const fruits: string[] = ["apple", "banana"];

// 末尾に追加
fruits.push("cherry");
console.log(fruits);  // ["apple", "banana", "cherry"]

// 末尾から削除
const last: string | undefined = fruits.pop();
console.log(last);    // "cherry"

// 先頭に追加
fruits.unshift("blueberry");
console.log(fruits);  // ["blueberry", "apple", "banana"]

// 先頭から削除
const first: string | undefined = fruits.shift();
console.log(first);   // "blueberry"
```

### スライス

```typescript
const arr: number[] = [0, 1, 2, 3, 4, 5];

// スライス（非破壊）
console.log(arr.slice(2, 4));    // [2, 3]
console.log(arr.slice(3));       // [3, 4, 5]
console.log(arr.slice(-2));      // [4, 5]

// スプライス（破壊的）
const removed: number[] = arr.splice(2, 2);  // インデックス2から2個削除
console.log(removed);   // [2, 3]
console.log(arr);       // [0, 1, 4, 5]
```

### ソート

```typescript
const numbers: number[] = [3, 1, 4, 1, 5, 9, 2];

// ソート（破壊的）
numbers.sort((a, b) => a - b);
console.log(numbers);  // [1, 1, 2, 3, 4, 5, 9]

// 逆順
numbers.sort((a, b) => b - a);

// 文字列ソート
const words: string[] = ["zebra", "apple", "banana"];
words.sort();  // アルファベット順
console.log(words);  // ["apple", "banana", "zebra"]
```

## 高階関数

### forEach

```typescript
const fruits: string[] = ["apple", "banana", "cherry"];

fruits.forEach((fruit: string, index: number) => {
    console.log(`${index}: ${fruit}`);
});
```

### map

```typescript
const numbers: number[] = [1, 2, 3, 4, 5];

// 各要素を2倍に
const doubled: number[] = numbers.map((n: number) => n * 2);
console.log(doubled);  // [2, 4, 6, 8, 10]

// 文字列に変換
const strings: string[] = numbers.map((n: number) => `Number: ${n}`);
console.log(strings);  // ["Number: 1", "Number: 2", ...]
```

### filter

```typescript
const numbers: number[] = [1, 2, 3, 4, 5, 6];

// 偶数だけ
const evens: number[] = numbers.filter((n: number) => n % 2 === 0);
console.log(evens);  // [2, 4, 6]

// 3以上
const large: number[] = numbers.filter((n: number) => n >= 3);
console.log(large);  // [3, 4, 5, 6]
```

### reduce

```typescript
const numbers: number[] = [1, 2, 3, 4, 5];

// 合計
const sum: number = numbers.reduce((acc: number, n: number) => acc + n, 0);
console.log(sum);  // 15

// 最大値
const max: number = numbers.reduce((max: number, n: number) => n > max ? n : max);
console.log(max);  // 5

// オブジェクトを作成
const words: string[] = ["apple", "banana", "apple"];
const count = words.reduce((acc: { [key: string]: number }, word: string) => {
    acc[word] = (acc[word] || 0) + 1;
    return acc;
}, {});
console.log(count);  // { apple: 2, banana: 1 }
```

### find・findIndex

```typescript
const users: { id: number; name: string }[] = [
    { id: 1, name: "Alice" },
    { id: 2, name: "Bob" },
    { id: 3, name: "Charlie" }
];

// 最初にマッチする要素
const user: { id: number; name: string } | undefined = 
    users.find((u: { id: number; name: string }) => u.id === 2);
console.log(user);  // { id: 2, name: "Bob" }

// インデックス
const index: number = users.findIndex((u: { id: number; name: string }) => u.id === 2);
console.log(index);  // 1
```

## 配列の検査

```typescript
const numbers: number[] = [1, 2, 3, 4, 5];

// 要素を含むか
console.log(numbers.includes(3));   // true
console.log(numbers.includes(10));  // false

// すべてが条件を満たすか
const allPositive: boolean = numbers.every((n: number) => n > 0);
console.log(allPositive);  // true

// いずれかが条件を満たすか
const hasEven: boolean = numbers.some((n: number) => n % 2 === 0);
console.log(hasEven);  // true
```

## スプレッド演算子

```typescript
const arr1: number[] = [1, 2, 3];
const arr2: number[] = [4, 5, 6];

// 配列の連結
const combined: number[] = [...arr1, ...arr2];
console.log(combined);  // [1, 2, 3, 4, 5, 6]

// 要素を挿入
const withMiddle: number[] = [arr1[0], 0, ...arr1.slice(1), ...arr2];
console.log(withMiddle);  // [1, 0, 2, 3, 4, 5, 6]

// コピー
const copy: number[] = [...arr1];
copy[0] = 100;
console.log(arr1);   // [1, 2, 3]（元は変わらない）
console.log(copy);   // [100, 2, 3]
```

## 多次元配列

```typescript
// 2次元配列
const matrix: number[][] = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
];

console.log(matrix[1][2]);  // 6

// 初期化
const grid: number[][] = Array(3).fill(null).map(() => Array(3).fill(0));
grid[1][1] = 5;
console.log(grid);
```

## 📌 重要なポイント

- **型指定**: `string[]`または`Array<string>`
- **破壊的メソッド**: `push`, `pop`, `sort`, `splice`は元の配列を変更
- **非破壊メソッド**: `map`, `filter`, `slice`は新しい配列を返す
- **高階関数**: `map`, `filter`, `reduce`で関数型プログラミング
- **スプレッド演算子**: 配列の展開とコピーに便利
