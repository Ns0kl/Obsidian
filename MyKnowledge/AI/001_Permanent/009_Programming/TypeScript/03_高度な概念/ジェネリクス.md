# ジェネリクス

## 概要

**ジェネリクス**は、型の安全性を保ちながら再利用可能なコンポーネントを作成できます。これはTypeScript特有の強力な機能です。

## 基本的なジェネリクス

### ジェネリック関数

```typescript
// 型パラメータT
function identity<T>(value: T): T {
    return value;
}

// 使用時に型を指定
const num: number = identity<number>(42);
const str: string = identity<string>("Hello");

// 型推論も可（<>を省略可）
const bool: boolean = identity(true);
```

### ジェネリック配列

```typescript
function getFirstElement<T>(arr: T[]): T {
    return arr[0];
}

const numbers: number = getFirstElement<number>([1, 2, 3]);
const words: string = getFirstElement<string>(["a", "b", "c"]);

// 型推論
const value = getFirstElement([10, 20, 30]);  // number型と推論
```

### ジェネリック型エイリアス

```typescript
type Container<T> = {
    value: T;
    isEmpty(): boolean;
};

const numberContainer: Container<number> = {
    value: 42,
    isEmpty() {
        return false;
    }
};

const stringContainer: Container<string> = {
    value: "Hello",
    isEmpty() {
        return this.value.length === 0;
    }
};
```

## ジェネリッククラス

```typescript
class Stack<T> {
    private items: T[] = [];

    push(item: T): void {
        this.items.push(item);
    }

    pop(): T | undefined {
        return this.items.pop();
    }

    peek(): T | undefined {
        return this.items[this.items.length - 1];
    }

    size(): number {
        return this.items.length;
    }
}

// 使用
const numberStack = new Stack<number>();
numberStack.push(1);
numberStack.push(2);
console.log(numberStack.pop());  // 2

const stringStack = new Stack<string>();
stringStack.push("Hello");
stringStack.push("World");
console.log(stringStack.peek());  // "World"
```

## ジェネリックインターフェース

```typescript
interface Repository<T> {
    getById(id: number): T;
    save(item: T): void;
    delete(id: number): void;
}

interface User {
    id: number;
    name: string;
}

class UserRepository implements Repository<User> {
    private users: User[] = [];

    getById(id: number): User | undefined {
        return this.users.find(u => u.id === id);
    }

    save(user: User): void {
        this.users.push(user);
    }

    delete(id: number): void {
        this.users = this.users.filter(u => u.id !== id);
    }
}
```

## 複数の型パラメータ

```typescript
function merge<T, U>(obj1: T, obj2: U): T & U {
    return { ...obj1, ...obj2 };
}

const merged = merge(
    { name: "Alice" },
    { age: 30 }
);

console.log(merged.name);  // "Alice"
console.log(merged.age);   // 30

// キーと値のペア
function createPair<K, V>(key: K, value: V): { key: K; value: V } {
    return { key, value };
}

const pair = createPair("name", "Alice");
console.log(pair.key);    // "name"
console.log(pair.value);  // "Alice"
```

## ジェネリック制約（Constraints）

```typescript
// Tは length プロパティを持つ型
function getLength<T extends { length: number }>(value: T): number {
    return value.length;
}

console.log(getLength("Hello"));     // 5
console.log(getLength([1, 2, 3]));   // 3
console.log(getLength({ length: 10 }));  // 10

// インターフェース制約
interface HasId {
    id: number;
}

function processItem<T extends HasId>(item: T): void {
    console.log(`ID: ${item.id}`);
}

processItem({ id: 1, name: "Alice" });  // OK
// processItem({ name: "Bob" });  // Error: id プロパティなし
```

### 型パラメータの制約

```typescript
// KeyofXを使用
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
    return obj[key];
}

const user = { name: "Alice", age: 30 };
const name = getProperty(user, "name");   // OK: string
// const email = getProperty(user, "email");  // Error: keyofで制限される
```

## 条件付き型

```typescript
// Tが string か確認
type IsString<T> = T extends string ? true : false;

type A = IsString<"hello">;      // true
type B = IsString<number>;       // false

// 実用例：戻り値の型を条件分岐
function processValue<T>(value: T): T extends string ? string : number {
    if (typeof value === "string") {
        return value.toUpperCase() as any;
    }
    return (value as number) * 2;
}

const result1: string = processValue("hello");  // string
const result2: number = processValue(42);       // number
```

## Mapped Types

```typescript
// すべてのプロパティをreadonlyに
type Readonly<T> = {
    readonly [K in keyof T]: T[K];
};

interface User {
    name: string;
    age: number;
}

type ReadonlyUser = Readonly<User>;
// { readonly name: string; readonly age: number; }

// すべてのプロパティをnullableに
type Nullable<T> = {
    [K in keyof T]: T[K] | null;
};

type NullableUser = Nullable<User>;
// { name: string | null; age: number | null; }
```

## 組み込みユーティリティ型

```typescript
interface User {
    id: number;
    name: string;
    email: string;
    age: number;
}

// Partial：すべてのプロパティをオプショナルに
type PartialUser = Partial<User>;
// { id?: number; name?: string; email?: string; age?: number; }

// Required：すべてのプロパティを必須に
type RequiredUser = Required<PartialUser>;
// { id: number; name: string; email: string; age: number; }

// Pick：指定されたプロパティだけを選択
type UserNameAndEmail = Pick<User, "name" | "email">;
// { name: string; email: string; }

// Omit：指定されたプロパティを除外
type UserWithoutId = Omit<User, "id">;
// { name: string; email: string; age: number; }

// Record：キーと値の型を指定
type UserRoles = Record<"admin" | "user" | "guest", User>;
// { admin: User; user: User; guest: User; }

// ReturnType：関数の戻り値型を取得
type GetName = (id: number) => string;
type NameType = ReturnType<GetName>;  // string
```

## ジェネリックのベストプラクティス

```typescript
// 1. 型パラメータに制約を設ける
function clone<T extends { clone(): T }>(obj: T): T {
    return obj.clone();
}

// 2. デフォルト型を指定
type Container<T = string> = {
    value: T;
};

const stringContainer: Container = { value: "Hello" };
const numberContainer: Container<number> = { value: 42 };

// 3. 複雑な型パラメータは名前を工夫
function processList<T, K extends keyof T>(list: T[], key: K): void {
    // 実装
}

// 4. 型の安全性を確認
function merge<T extends object, U extends object>(
    obj1: T,
    obj2: U
): T & U {
    return { ...obj1, ...obj2 };
}
```

```typescript
interface PaginatedResult<T> {
    items: T[];
    total: number;

async function getUsers(): Promise<ApiResponse<PaginatedResult<User>>> {
    // ...
}

    const response = await window.fetch(url);
    return response.json();
}

const user: User = await fetch("/api/user/1");
```

## 📌 重要なポイント

- **ジェネリック**: 再利用可能で型安全なコンポーネント
- **型パラメータ**: `<T>`で汎用的な型を指定
- **制約**: `extends`で型パラメータを制限
- **Mapped Types**: プロパティを一括変換
- **ユーティリティ型**: Partial, Required, Pick, Omit など
- **条件付き型**: 型を条件分岐
