# 非同期処理

## 概要

**非同期処理**により、長時間かかる操作（API呼び出し、ファイル読み込みなど）を実行する際、プログラムがブロックされません。

## コールバック

### 基本形

```typescript
function fetchData(callback: (data: string) => void): void {
    setTimeout(() => {
        callback("データが取得されました");
    }, 1000);
}

fetchData((data: string) => {
    console.log(data);  // 1秒後に出力
});

console.log("処理が続く");
```

### エラーハンドリング

```typescript
function divide(
    a: number,
    b: number,
    callback: (error: Error | null, result?: number) => void
): void {
    if (b === 0) {
        callback(new Error("ゼロで除算できません"));
    } else {
        callback(null, a / b);
    }
}

divide(10, 2, (error, result) => {
    if (error) {
        console.log("エラー:", error.message);
    } else {
        console.log("結果:", result);
    }
});
```

### Callback Hell

```typescript
// ネストが深くなる（非推奨）
function getUser(id: number, callback: (user: any) => void): void {
    setTimeout(() => callback({ id, name: "Alice" }), 100);
}

function getOrders(userId: number, callback: (orders: any[]) => void): void {
    setTimeout(() => callback([{ id: 1, amount: 100 }]), 100);
}

getUser(1, (user) => {
    console.log(user);
    getOrders(user.id, (orders) => {
        console.log(orders);
        // さらにネストが続く...
    });
});
```

## Promise

### 基本形

```typescript
function fetchData(url: string): Promise<string> {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            if (url) {
                resolve("データが取得されました");
            } else {
                reject(new Error("URLが無効です"));
            }
        }, 1000);
    });
}

// .then()と.catch()
fetchData("https://api.example.com")
    .then((data: string) => {
        console.log("成功:", data);
    })
    .catch((error: Error) => {
        console.log("エラー:", error.message);
    });
```

### then チェーン

```typescript
function step1(): Promise<number> {
    return new Promise(resolve => {
        setTimeout(() => resolve(1), 100);
    });
}

function step2(input: number): Promise<number> {
    return new Promise(resolve => {
        setTimeout(() => resolve(input * 2), 100);
    });
}

function step3(input: number): Promise<number> {
    return new Promise(resolve => {
        setTimeout(() => resolve(input + 10), 100);
    });
}

step1()
    .then((result: number) => {
        console.log(result);  // 1
        return step2(result);
    })
    .then((result: number) => {
        console.log(result);  // 2
        return step3(result);
    })
    .then((result: number) => {
        console.log(result);  // 12
    })
    .catch((error: Error) => {
        console.log("エラー:", error);
    });
```

### Promise.all

```typescript
const promise1: Promise<string> = Promise.resolve("1");
const promise2: Promise<string> = Promise.resolve("2");
const promise3: Promise<string> = Promise.resolve("3");

// すべてのPromiseが完了を待つ
Promise.all([promise1, promise2, promise3])
    .then((results: string[]) => {
        console.log(results);  // ["1", "2", "3"]
    });

// 複数のAPI呼び出し
function fetchUser(id: number): Promise<{ id: number; name: string }> {
    return Promise.resolve({ id, name: `User${id}` });
}

Promise.all([
    fetchUser(1),
    fetchUser(2),
    fetchUser(3)
])
    .then((users) => {
        console.log(users);
    });
```

### Promise.race

```typescript
const promise1: Promise<string> = new Promise(resolve =>
    setTimeout(() => resolve("1秒後"), 1000)
);

const promise2: Promise<string> = new Promise(resolve =>
    setTimeout(() => resolve("2秒後"), 2000)
);

// 最初に完了したPromiseの結果を使う
Promise.race([promise1, promise2])
    .then((result: string) => {
        console.log(result);  // "1秒後"
    });
```

### Promise.allSettled

```typescript
const promises: Promise<any>[] = [
    Promise.resolve("成功"),
    Promise.reject(new Error("失敗")),
    Promise.resolve("成功2")
];

Promise.allSettled(promises)
    .then((results) => {
        results.forEach((result) => {
            if (result.status === "fulfilled") {
                console.log("成功:", result.value);
            } else {
                console.log("失敗:", result.reason);
            }
        });
    });
```

## async・await

### 基本形

```typescript
function delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
}

async function myAsyncFunction(): Promise<void> {
    console.log("開始");
    await delay(1000);
    console.log("1秒後");
    await delay(1000);
    console.log("2秒後");
}

myAsyncFunction();
```

### 戻り値

```typescript
async function fetchUser(id: number): Promise<{ id: number; name: string }> {
    // 実際の実装ではfetchを使用
    return new Promise(resolve => {
        setTimeout(() => {
            resolve({ id, name: "Alice" });
        }, 1000);
    });
}

// 呼び出し
(async () => {
    const user = await fetchUser(1);
    console.log(user.name);  // "Alice"
})();
```

### エラーハンドリング

```typescript
async function safeFetch(url: string): Promise<string> {
    try {
        const response: Response = await fetch(url);
        if (!response.ok) {
            throw new Error(`HTTP ${response.status}`);
        }
        return await response.text();
    } catch (error) {
        if (error instanceof Error) {
            console.log("エラー:", error.message);
        }
        throw error;
    }
}

// 呼び出し側
(async () => {
    try {
        const data: string = await safeFetch("https://api.example.com/data");
        console.log(data);
    } catch (error) {
        console.log("最終的なエラー処理");
    }
})();
```

### 複数のawait

```typescript
async function getUserWithOrders(userId: number): Promise<void> {
    const user = await fetchUser(userId);
    console.log("ユーザー:", user);

    const orders = await fetchOrders(userId);
    console.log("注文:", orders);
}

// 並列実行（同時に実行）
async function getDataInParallel(userId: number): Promise<void> {
    const [user, orders] = await Promise.all([
        fetchUser(userId),
        fetchOrders(userId)
    ]);

    console.log("ユーザー:", user);
    console.log("注文:", orders);
}
```

## リトライロジック

```typescript
async function fetchWithRetry(
    url: string,
    maxRetries: number = 3,
    delay: number = 1000
): Promise<Response> {
    for (let i = 0; i < maxRetries; i++) {
        try {
            return await fetch(url);
        } catch (error) {
            if (i < maxRetries - 1) {
                console.log(`リトライ ${i + 1}/${maxRetries - 1}`);
                await new Promise(resolve => setTimeout(resolve, delay));
            } else {
                throw error;
            }
        }
    }
    throw new Error("All retries exhausted");
}

(async () => {
    try {
        const response = await fetchWithRetry("https://api.example.com");
        console.log(response);
    } catch (error) {
        console.log("最終的に失敗");
    }
})();
```

## タイムアウト

```typescript
function withTimeout<T>(
    promise: Promise<T>,
    timeoutMs: number
): Promise<T> {
    return Promise.race([
        promise,
        new Promise<T>((_, reject) =>
            setTimeout(() => reject(new Error("Timeout")), timeoutMs)
        )
    ]);
}

async function slowFetch(url: string): Promise<string> {
    const response = await withTimeout(fetch(url), 5000);
    return response.text();
}

(async () => {
    try {
        const data = await slowFetch("https://api.example.com");
        console.log(data);
    } catch (error) {
        if (error instanceof Error && error.message === "Timeout") {
            console.log("タイムアウト");
        }
    }
})();
```

## 📌 重要なポイント

- **コールバック**: シンプルだがネストが深くなる可能性
- **Promise**: チェーンが可能で読みやすい
- **async/await**: 同期的に見えるコード構造
- **Promise.all**: 複数のPromiseを並列実行
- **エラーハンドリング**: try/catchでPromiseのエラーを処理
- **リトライ・タイムアウト**: 本番環境では重要な技術
