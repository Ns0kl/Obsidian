# アルゴリズムと数値計算

## 概要

プログラミングの実践において、基本的なアルゴリズムと数値計算技法を理解することは重要です。

## ソートアルゴリズム

### バブルソート

```python
def bubble_sort(arr):
    n = len(arr)
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                # スワップ
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr

arr = [64, 34, 25, 12, 22, 11, 90]
result = bubble_sort(arr.copy())
print(result)  # [11, 12, 22, 25, 34, 64, 90]
```

### Pythonの標準ソート

```python
# sorted()関数（非破壊）
arr = [64, 34, 25, 12, 22, 11, 90]
sorted_arr = sorted(arr)
print(sorted_arr)  # [11, 12, 22, 25, 34, 64, 90]

# sort()メソッド（破壊的）
arr.sort()
print(arr)  # [11, 12, 22, 25, 34, 64, 90]

# 降順
sorted_desc = sorted(arr, reverse=True)
print(sorted_desc)  # [90, 64, 34, 25, 22, 12, 11]

# キー関数を使用
words = ['apple', 'pie', 'zoo', 'cat']
sorted_by_len = sorted(words, key=len)
print(sorted_by_len)  # ['pie', 'cat', 'zoo', 'apple']
```

## 探索アルゴリズム

### 線形探索

```python
def linear_search(arr, target):
    for i, value in enumerate(arr):
        if value == target:
            return i
    return -1

arr = [10, 20, 30, 40, 50]
idx = linear_search(arr, 30)
print(idx)  # 2
```

### 二分探索

```python
def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    
    while left <= right:
        mid = (left + right) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    
    return -1

arr = [10, 20, 30, 40, 50]
idx = binary_search(arr, 30)
print(idx)  # 2
```

## 数値計算

### 最大公約数（GCD）

```python
def gcd(a, b):
    while b:
        a, b = b, a % b
    return a

print(gcd(48, 18))  # 6

# Pythonの標準ライブラリ
import math
print(math.gcd(48, 18))  # 6
```

### 最小公倍数（LCM）

```python
def lcm(a, b):
    import math
    return abs(a * b) // math.gcd(a, b)

print(lcm(12, 18))  # 36
```

### 階乗

```python
def factorial(n):
    if n == 0:
        return 1
    return n * factorial(n - 1)

print(factorial(5))  # 120

# 標準ライブラリ
import math
print(math.factorial(5))  # 120
```

### フィボナッチ数列

```python
# 再帰的（遅い）
def fibonacci_recursive(n):
    if n <= 1:
        return n
    return fibonacci_recursive(n - 1) + fibonacci_recursive(n - 2)

print(fibonacci_recursive(10))  # 55

# 動的計画法（効率的）
def fibonacci_dp(n):
    if n <= 1:
        return n
    a, b = 0, 1
    for _ in range(2, n + 1):
        a, b = b, a + b
    return b

print(fibonacci_dp(10))  # 55
```

### 素数判定

```python
def is_prime(n):
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    
    for i in range(3, int(n**0.5) + 1, 2):
        if n % i == 0:
            return False
    return True

print(is_prime(17))  # True
print(is_prime(18))  # False
```

## 数値計算手法

### Newton-Raphson法（平方根計算）

```python
def sqrt_newton(x, tolerance=1e-10):
    '''Newton-Raphson法で平方根を計算'''
    if x < 0:
        raise ValueError('負の数の平方根は計算できません')
    
    guess = x / 2.0  # 初期推測
    while True:
        # f(x) = x^2 - 2のグラフでニュートン法を適用
        new_guess = (guess + x / guess) / 2.0
        
        if abs(new_guess - guess) < tolerance:
            return new_guess
        
        guess = new_guess

print(sqrt_newton(2))   # 1.414213562373095
print(sqrt_newton(9))   # 3.0
```

### 二分探索法（方程式の解を求める）

```python
def bisection(f, a, b, tolerance=1e-6):
    '''区間[a, b]で方程式f(x)=0の解を二分探索で求める'''
    if f(a) * f(b) >= 0:
        raise ValueError('f(a)とf(b)の符号が異なる必要があります')
    
    while abs(b - a) > tolerance:
        c = (a + b) / 2
        if f(c) == 0:
            return c
        elif f(a) * f(c) < 0:
            b = c
        else:
            a = c
    
    return (a + b) / 2

# x^2 - 2 = 0の解を求める
f = lambda x: x**2 - 2
root = bisection(f, 1, 2)
print(root)  # 1.414213562011719
```

## 統計計算

```python
# 平均
data = [1, 2, 3, 4, 5]
mean = sum(data) / len(data)
print(mean)  # 3.0

# 中央値
sorted_data = sorted(data)
n = len(sorted_data)
median = sorted_data[n // 2] if n % 2 == 1 else (sorted_data[n // 2 - 1] + sorted_data[n // 2]) / 2
print(median)  # 3

# 標準偏差
import statistics
std_dev = statistics.stdev(data)
print(std_dev)  # 1.5811388300841898

# 分散
variance = statistics.variance(data)
print(variance)  # 2.5
```

## 📌 重要なポイント

- **組み込みソート**: `sorted()`と`.sort()`は常にO(n log n)
- **二分探索**: ソート済み配列では効率的
- **動的計画法**: 再帰的な問題を効率的に解く
- **数値精度**: 浮動小数点演算の精度に注意
- **標準ライブラリ**: `math`や`statistics`モジュールを活用
